function(rr_recipe script input output)
  execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/${script}"
    "${PSXRECOMP_ROOT}/runtime/src/${input}" "${CMAKE_CURRENT_BINARY_DIR}/${output}"
    COMMAND_ERROR_IS_FATAL ANY)
  target_sources(rr-game PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/${output}")
endfunction()

# Speed profile (PGO, +17-36% measured): recipe/game.profdata, trained on this very
# build layout (tools/pgo_train_public.py), so its function names are the relative
# compile paths every install uses. It holds function names, control-flow hashes and
# execution counts of the compiled code only: no game code or data. RR_PGO_GENERATE
# builds the instrumented trainer instead. Set before the fast-timing library copies
# rr-game's options, so the generated game code gets the same profile.
option(RR_PGO_GENERATE "Instrumented build for training the speed profile" OFF)
set(RR_PGO_USE "" CACHE FILEPATH "Speed profile (.profdata); empty = none")
if(RR_PGO_GENERATE)
  target_compile_options(rr-game PRIVATE -fprofile-generate)
  target_link_options(rr-game PRIVATE -fprofile-generate)
elseif(RR_PGO_USE AND EXISTS "${RR_PGO_USE}")
  target_compile_options(rr-game PRIVATE "-fprofile-use=${RR_PGO_USE}"
    -Wno-profile-instr-unprofiled -Wno-profile-instr-out-of-date -Wno-backend-plugin)
  # A new profile at the same path must recompile everything (objects built with
  # different profiles cannot be linked together): its hash is in every command.
  file(SHA256 "${RR_PGO_USE}" _pgo_hash)
  string(SUBSTRING "${_pgo_hash}" 0 16 _pgo_hash)
  target_compile_definitions(rr-game PRIVATE "RR_PGO_PROFILE=0x${_pgo_hash}")
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${RR_PGO_USE}")
  # The linker's call-graph function sort (fed by the profile) orders code
  # differently on every link; without it every player's build is identical.
  target_link_options(rr-game PRIVATE "-Wl,-Xlink=-call-graph-profile-sort:no")
endif()

# Reproducible builds, part 2: compile with source paths relative to the build
# folder (tools/relpath_cc.py, in front of ccache), so ThinLTO's names for
# shared private symbols, and with them the data layout, are the same in every
# install. The prefix maps above cover __FILE__; this covers the module itself.
foreach(_lang C CXX)
  get_property(_next TARGET rr-game PROPERTY ${_lang}_COMPILER_LAUNCHER)
  set_property(TARGET rr-game rr-fast-gen PROPERTY ${_lang}_COMPILER_LAUNCHER
    "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/relpath_cc.py" ${_next})
  # Part 3: the link sees every input under a numbered name in the build folder (tools/relpath_ld.py).
  set_property(TARGET rr-game PROPERTY ${_lang}_LINKER_LAUNCHER
    "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/relpath_ld.py")
endforeach()
# Compiler settings (0.15 compiler pass). Baseline x86-64, no -march: building
# for the player's own CPU measured no gain (with or without LTO) and would stop
# a copied installation from running on an older CPU. No fast-math or other
# relaxed floating-point/integer flags: online play needs two PCs to simulate
# bit-identically. PGO is not used here (see PERFORMANCE.md).

# ThinLTO over the runtime and the generated game code (+11% emulation
# throughput on the test PC). RR_LINK_JOBS backend jobs (install_game.py picks
# them from the PC's threads and memory, about 1 GB each); the output is the same
# for any number. Falls back to a normal link if the compiler or linker cannot do it.
# The --wrap hooks below keep working: lld stops LTO from inlining wrapped symbols.
option(RR_THINLTO "Link the runtime with ThinLTO" ON)
set(RR_LINK_JOBS 2 CACHE STRING "ThinLTO backend jobs")
if(RR_THINLTO)
  include(CheckCSourceCompiles)
  set(CMAKE_REQUIRED_FLAGS "-flto=thin")
  set(CMAKE_REQUIRED_LINK_OPTIONS "-flto=thin" "-Wl,--thinlto-jobs=${RR_LINK_JOBS}" "-Wl,--lto-O3")
  check_c_source_compiles("int rr_lto_probe(int x) { return x * 3; } int main(void) { return rr_lto_probe(0); }" RR_THINLTO_LINKS)
  unset(CMAKE_REQUIRED_FLAGS)
  unset(CMAKE_REQUIRED_LINK_OPTIONS)
  if(RR_THINLTO_LINKS)
    target_compile_options(rr-game PRIVATE -flto=thin)
    target_compile_options(rr-fast-gen PRIVATE -flto=thin)
    target_link_options(rr-game PRIVATE -flto=thin "-Wl,--thinlto-jobs=${RR_LINK_JOBS}" -Wl,--lto-O3
      "-Wl,--thinlto-cache-dir=${CMAKE_CURRENT_BINARY_DIR}/thinlto-cache")
  else()
    message(WARNING "ThinLTO is not available with this compiler/linker; linking without it")
  endif()
endif()
